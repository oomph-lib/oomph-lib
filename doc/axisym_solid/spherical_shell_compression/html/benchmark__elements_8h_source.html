37 /// A class for elements to solve the 1D radial displacement ODE
38 /// in the spherical shell compression benchmark problem.
39 //=============================================================
40 template<unsigned N_NODE1D>
41 class ShellBenchmarkElement : public virtual QElement<1, N_NODE1D>,
42 public virtual FiniteElement
43 {
44 public:
45 /// Constructor ()
46 ShellBenchmarkElement() : Nu_pt(0) {}
47
48 /// Broken copy constructor
49 ShellBenchmarkElement(const ShellBenchmarkElement& dummy) = delete;
50
51 /// Broken assignment operator
52 void operator=(const ShellBenchmarkElement&) = delete;
53
54 /// Required # of `values' (pinned or dofs) at node n
55 inline unsigned required_nvalue(const unsigned& n) const
56 {
57 return 1;
58 }
59
60 /// Access function: Poisson ratio
61 const double& nu() const
62 {
63 return *Nu_pt;
64 }
65
66 /// Pointer to Poisson ratio
67 double*& nu_pt()
68 {
69 return Nu_pt;
70 }
71
72 /// Output FE representation of soln: x,y,u or x,y,z,u at
73 /// n_plot^DIM plot points
74 void output(std::ostream& outfile, const unsigned& n_plot);
75
76 /// Add the element's contribution to its residual vector (wrapper)
77 void fill_in_contribution_to_residuals(Vector<double>& residuals)
78 {
79 // Call the generic residuals function with flag set to 0
80 // using a dummy matrix argument
81 fill_in_generic_residual_contribution_shell_benchmark(residuals);
82 }
83
84 /// Return FE representation of function value u at local coordinate s
85 virtual inline double interpolated_u_shell_benchmark(
86 const Vector<double>& s) const
87 {
88 // Find number of nodes
89 const unsigned n_node = nnode();
90
91 // Get the index at which the poisson unknown is stored
92 const unsigned u_nodal_index = 0; // u_index_shell_benchmark();
93
94 // Local shape function
95 Shape psi(n_node);
96
97 // Find values of shape function
98 shape(s, psi);
99
100 // Initialise value of u
101 double interpolated_u = 0.0;
102
103 // Loop over the local nodes and sum
104 for (unsigned l = 0; l < n_node; l++)
105 {
106 interpolated_u += this->nodal_value(l, u_nodal_index) * psi[l];
107 }
108
109 return (interpolated_u);
110 }
111
112 protected:
113 /// Compute element residual Vector only (if flag=and/or element
114 /// Jacobian matrix
115 virtual void fill_in_generic_residual_contribution_shell_benchmark(
116 Vector<double>& residuals);
117
118 /// Pointer to the Poisson ratio
119 double* Nu_pt;
120 };

126 class ShellBenchmarkPressureElement : public virtual PointElement,
